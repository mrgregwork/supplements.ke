<script define:vars={{ subcategories: JSON.stringify(subcategories) }}>
  const parsedSubcategories = JSON.parse(subcategories);
  
  const searchInput = document.getElementById('search-query') as HTMLInputElement;
  const searchBtn = document.getElementById('search-btn');
  const searchStatus = document.getElementById('search-status');
  const searchResults = document.getElementById('search-results');
  const importPanel = document.getElementById('import-panel');
  const categorySelect = document.getElementById('import-category') as HTMLSelectElement;
  const subcategorySelect = document.getElementById('import-subcategory') as HTMLSelectElement;
  const importBtn = document.getElementById('import-btn');
  const cancelImportBtn = document.getElementById('cancel-import-btn');
  const selectedProductPreview = document.getElementById('selected-product-preview');
  const importPriceInput = document.getElementById('import-price') as HTMLInputElement;
  const importDescInput = document.getElementById('import-description') as HTMLTextAreaElement;
  
  let selectedProduct: any = null;
  
  categorySelect?.addEventListener('change', () => {
    const categoryId = categorySelect.value;
    const filtered = parsedSubcategories.filter((s: any) => s.categoryId === categoryId);
    
    subcategorySelect.innerHTML = '<option value="">Select a subcategory</option>';
    filtered.forEach((sub: any) => {
      const option = document.createElement('option');
      option.value = String(sub.id);
      option.textContent = sub.name;
      subcategorySelect.appendChild(option);
    });
  });
  
  searchBtn?.addEventListener('click', async () => {
    const query = searchInput?.value?.trim();
    if (!query) {
      alert('Please enter a search term');
      return;
    }
    
    if (searchStatus) {
      searchStatus.textContent = 'Searching...';
      searchStatus.classList.remove('hidden');
    }
    
    try {
      const response = await fetch('/api/admin/import/search', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ query }),
      });
      
      const result = await response.json();
      
      if (!response.ok) {
        throw new Error(result.message || 'Search failed');
      }
      
      if (searchStatus) searchStatus.classList.add('hidden');
      
      if (result.products && result.products.length > 0) {
        renderSearchResults(result.products);
      } else {
        if (searchResults) {
          searchResults.innerHTML = `
            <p class="text-muted-foreground text-center py-8">
              No products found for "${query}". Try a different search term.
            </p>
          `;
        }
      }
    } catch (error: any) {
      if (searchStatus) {
        searchStatus.textContent = error.message || 'Search failed. Please try again.';
        searchStatus.classList.remove('hidden');
      }
    }
  });
  
  searchInput?.addEventListener('keypress', (e) => {
    if (e.key === 'Enter') {
      searchBtn?.click();
    }
  });
  
  function renderSearchResults(products: any[]) {
    if (!searchResults) return;
    
    searchResults.innerHTML = products.map((product, index) => `
      <div class="border rounded-lg p-4 hover:bg-muted/50 transition cursor-pointer" data-product-index="${index}" data-testid="product-result-${index}">
        <div class="flex gap-4">
          <div class="w-20 h-20 bg-muted rounded-md flex-shrink-0 overflow-hidden">
            ${product.image ? `<img src="${product.image}" alt="${product.title}" class="w-full h-full object-cover" />` : ''}
          </div>
          <div class="flex-1 min-w-0">
            <h4 class="font-medium truncate">${product.title}</h4>
            <p class="text-sm text-muted-foreground line-clamp-2">${product.description || ''}</p>
            <div class="flex items-center gap-4 mt-2">
              <span class="font-semibold">${product.currency || '$'}${product.price || 'N/A'}</span>
              ${product.rating ? `<span class="text-sm text-muted-foreground">${product.rating} stars</span>` : ''}
            </div>
          </div>
          <button 
            class="select-product-btn px-4 py-2 bg-primary text-primary-foreground rounded-md hover:opacity-90 transition self-center"
            data-product='${JSON.stringify(product)}'
            data-testid="button-select-product-${index}"
          >
            Select
          </button>
        </div>
      </div>
    `).join('');
    
    document.querySelectorAll('.select-product-btn').forEach(btn => {
      btn.addEventListener('click', (e) => {
        e.stopPropagation();
        const productData = (btn as HTMLElement).dataset.product;
        if (productData) {
          selectProduct(JSON.parse(productData));
        }
      });
    });
  }
  
  function selectProduct(product: any) {
    selectedProduct = product;
    
    if (selectedProductPreview) {
      selectedProductPreview.innerHTML = `
        <div class="border rounded-lg p-4">
          <div class="w-full h-48 bg-muted rounded-md mb-4 overflow-hidden">
            ${product.image ? `<img src="${product.image}" alt="${product.title}" class="w-full h-full object-cover" />` : ''}
          </div>
          <h4 class="font-semibold">${product.title}</h4>
          <p class="text-lg font-bold mt-2">${product.currency || '$'}${product.price || 'N/A'}</p>
          ${product.rating ? `<p class="text-sm text-muted-foreground">${product.rating} stars</p>` : ''}
        </div>
      `;
    }
    
    if (importPriceInput) importPriceInput.value = product.price || '';
    if (importDescInput) importDescInput.value = product.description || '';
    
    importPanel?.classList.remove('hidden');
    importPanel?.scrollIntoView({ behavior: 'smooth' });
  }
  
  cancelImportBtn?.addEventListener('click', () => {
    selectedProduct = null;
    importPanel?.classList.add('hidden');
  });
  
  importBtn?.addEventListener('click', async () => {
    if (!selectedProduct) return;
    
    const categoryId = categorySelect?.value;
    const subcategoryId = subcategorySelect?.value;
    
    if (!categoryId || !subcategoryId) {
      alert('Please select a category and subcategory');
      return;
    }
    
    const importData = {
      title: selectedProduct.title,
      description: importDescInput?.value || selectedProduct.description || '',
      price: parseFloat(importPriceInput?.value) || selectedProduct.price || 0,
      image: selectedProduct.image || '',
      categoryId: categoryId,
      subcategoryId: subcategoryId,
      attributes: selectedProduct.attributes || {},
      sourceData: selectedProduct,
    };
    
    importBtn.textContent = 'Importing...';
    (importBtn as HTMLButtonElement).disabled = true;
    
    try {
      const response = await fetch('/api/admin/import/product', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(importData),
      });
      
      const result = await response.json();
      
      if (!response.ok) {
        throw new Error(result.message || 'Import failed');
      }
      
      alert('Product imported successfully!');
      selectedProduct = null;
      importPanel?.classList.add('hidden');
      
      if (result.productId) {
        window.location.href = `/admin/products?highlight=${result.productId}`;
      }
    } catch (error: any) {
      alert(error.message || 'Failed to import product');
    } finally {
      importBtn.textContent = 'Import Product';
      (importBtn as HTMLButtonElement).disabled = false;
    }
  });
</script>
